from datetime import datetime

def cron_to_win_task(cron_expr):
    try:
        # Dividir la expresión cron en sus componentes
        components = cron_expr.split()
        if len(components) < 5:
            raise ValueError("Expresión cron inválida: se requieren al menos 5 campos")

        minute, hour, day_of_month, month, day_of_week = components[:5]
        year = components[5] if len(components) > 5 else "*"

        # Convertir minutos
        if minute == "*":
            start_minute, interval_minutes = "0", "1"
        elif minute.startswith("*/"):
            start_minute, interval_minutes = "0", minute[2:]
        else:
            start_minute, interval_minutes = minute, "1"

        # Convertir horas
        if hour == "*":
            start_hour, interval_hours = "0", "1"
        elif hour.startswith("*/"):
            start_hour, interval_hours = "0", hour[2:]
        else:
            start_hour, interval_hours = hour, "1"

        # Convertir días de la semana
        days_of_week = None if day_of_week == "*" else day_of_week.replace("-", ",").split(",")

        # Convertir días del mes
        if day_of_month == "*":
            days_of_month = None
        elif day_of_month.startswith("*/"):
            days_of_month = [str(i) for i in range(1, 32, int(day_of_month[2:]))]
        else:
            days_of_month = day_of_month.split(",")

        # Convertir meses
        if month == "*":
            months = None
        elif month.startswith("*/"):
            months = [str(i) for i in range(1, 13, int(month[2:]))]
        else:
            months = month.split(",")

        # Crear la estructura de la tarea para Windows
        task_trigger = {
            "type": "daily",
            "start_boundary": datetime.now().strftime("%Y-%m-%dT%H:%M:%S"),
            "enabled": True,
            "schedule": {
                "repeat_every": f"PT{interval_hours}H{interval_minutes}M",
                "start_time": f"{start_hour}:{start_minute}:00",
            },
        }

        if days_of_week:
            task_trigger["type"] = "weekly"
            task_trigger["schedule"]["days_of_week"] = days_of_week

        if days_of_month:
            task_trigger["type"] = "monthly"
            task_trigger["schedule"]["days_of_month"] = days_of_month

        if months:
            task_trigger["schedule"]["months"] = months

        if year != "*":
            task_trigger["schedule"]["years"] = [year]

        return task_trigger

    except Exception as e:
        return {"error": str(e)}

class FilterModule(object):
    def filters(self):
        return {
            'cron_to_win_task': cron_to_win_task
        }
